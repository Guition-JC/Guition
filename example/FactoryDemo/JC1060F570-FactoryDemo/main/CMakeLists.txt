# Select the asset set that matches the board selected by ESP Board Manager.
# The SPIFFS image is still mounted at /assets at runtime, so application
# paths remain resolution-independent (for example, A:/assets/icon/img_wifi.png).
if(CONFIG_ESP_BOARD_JC1060F570C)
    set(ASSETS_SPIFFS_DIR ../spiffs/1024x600)
    message(STATUS "SPIFFS assets: 1024x600 (JC1060F570C)")
elseif(CONFIG_ESP_BOARD_JC8048F550)
    set(ASSETS_SPIFFS_DIR ../spiffs/800x480)
    message(STATUS "SPIFFS assets: 800x480 (JC8048F550)")
else()
    # A fresh worktree has no sdkconfig yet, so CONFIG_ESP_BOARD_* stays undefined
    # until `idf.py bmgr -b jc1060f570c` writes it. Default to the 1024x600 set
    # instead of aborting the very configure step bmgr needs.
    set(ASSETS_SPIFFS_DIR ../spiffs/1024x600)
    message(WARNING
        "No board selected yet; defaulting SPIFFS assets to 1024x600. "
        "Run `idf.py bmgr -b jc1060f570c` to select the board.")
endif()

spiffs_create_partition_image(assets ${ASSETS_SPIFFS_DIR} FLASH_IN_PROJECT)

# The bundled font is packed into its own partition (see partitions.csv) and
# memory-mapped at runtime: FreeType then reads glyphs straight out of the mapped
# image instead of streaming them through a file system. The generated header
# goes to the build directory, and the font's C identifier-safe file name keeps
# the generated enum valid.
spiffs_create_partition_assets(fonts ${CMAKE_CURRENT_LIST_DIR}/../fonts/embedded
    FLASH_IN_PROJECT
    MMAP_FILE_SUPPORT_FORMAT ".ttf"
    IMPORT_INC_PATH "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(${COMPONENT_LIB} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
# The header is generated by the assets target, so main must wait for it.
add_dependencies(${COMPONENT_LIB} assets_fonts_bin)

# esp_bt_audio's GMF I/O source includes gmf_core headers, but the installed
# component manifest does not expose that include path to its own target.
if(CONFIG_ESP_BT_AUDIO_GMF_IO_SUPPORT)
    idf_component_get_property(ESP_BT_AUDIO_COMPONENT_LIB espressif__esp_bt_audio COMPONENT_LIB)
    idf_component_get_property(GMF_CORE_COMPONENT_DIR espressif__gmf_core COMPONENT_DIR)
    target_include_directories(${ESP_BT_AUDIO_COMPONENT_LIB} PRIVATE
                               "${GMF_CORE_COMPONENT_DIR}/include")
endif()
